-- ========================================= -- دراج — إعداد قاعدة بيانات Supabase -- نفّذ هذا الكود كاملاً في SQL Editor في Supabase -- ========================================= -- 1. جدول المستخدمين (عملاء + مندوبون + محلات) CREATE TABLE IF NOT EXISTS drj_users ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, name text NOT NULL, phone text UNIQUE NOT NULL, pin text NOT NULL, role text NOT NULL CHECK (role IN ('client','driver','store','admin')), store_id text, photo text, created_at timestamptz DEFAULT now() ); -- 2. جدول المتاجر CREATE TABLE IF NOT EXISTS drj_stores ( id text PRIMARY KEY, name text NOT NULL, icon text DEFAULT '🏪', desc text, location text, phone text, pin text DEFAULT '0000', photo text, store_lat double precision, store_lng double precision, store_address text, is_open boolean DEFAULT true, cats jsonb DEFAULT '[]'::jsonb, created_at timestamptz DEFAULT now() ); -- 3. جدول الطلبات CREATE TABLE IF NOT EXISTS drj_orders ( id text PRIMARY KEY, store text, store_id text, client text, client_phone text, items text, subtotal numeric, status text DEFAULT 'new' CHECK (status IN ('new','confirmed','delivering','done','rejected')), time text, created_at timestamptz DEFAULT now(), driver text, driver_phone text, delivery_ms bigint, client_lat double precision, client_lng double precision, client_address text, done_at timestamptz ); -- 4. جدول الإعلانات CREATE TABLE IF NOT EXISTS drj_ads ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, img text, title text, sub text, sort_order int DEFAULT 0, created_at timestamptz DEFAULT now() ); -- 5. جدول العروض CREATE TABLE IF NOT EXISTS drj_promos ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, store_id text, title text, desc text, price numeric, img text, expiry date, created_at timestamptz DEFAULT now() ); -- 6. جدول التواجد اللحظي CREATE TABLE IF NOT EXISTS drj_online ( phone text PRIMARY KEY, name text, role text, store_id text, updated_at timestamptz DEFAULT now() ); -- ===== ROW LEVEL SECURITY ===== -- تفعيل RLS (مهم للأمان) ALTER TABLE drj_users ENABLE ROW LEVEL SECURITY; ALTER TABLE drj_stores ENABLE ROW LEVEL SECURITY; ALTER TABLE drj_orders ENABLE ROW LEVEL SECURITY; ALTER TABLE drj_ads ENABLE ROW LEVEL SECURITY; ALTER TABLE drj_promos ENABLE ROW LEVEL SECURITY; ALTER TABLE drj_online ENABLE ROW LEVEL SECURITY; -- سماح لـ anon بالقراءة والكتابة (التطبيق يتحكم في الصلاحيات بنفسه) CREATE POLICY "allow_all_drj_users" ON drj_users FOR ALL TO anon USING (true) WITH CHECK (true); CREATE POLICY "allow_all_drj_stores" ON drj_stores FOR ALL TO anon USING (true) WITH CHECK (true); CREATE POLICY "allow_all_drj_orders" ON drj_orders FOR ALL TO anon USING (true) WITH CHECK (true); CREATE POLICY "allow_all_drj_ads" ON drj_ads FOR ALL TO anon USING (true) WITH CHECK (true); CREATE POLICY "allow_all_drj_promos" ON drj_promos FOR ALL TO anon USING (true) WITH CHECK (true); CREATE POLICY "allow_all_drj_online" ON drj_online FOR ALL TO anon USING (true) WITH CHECK (true); -- ===== REALTIME ===== -- تفعيل Realtime على جدول الطلبات والتواجد ALTER PUBLICATION supabase_realtime ADD TABLE drj_orders; ALTER PUBLICATION supabase_realtime ADD TABLE drj_online; -- =====
